{{- /* A card's preview in a `view: cards` folder (layouts/folder.html): { src, width, height, latest }, from a data item and a Fill size such as "320x180". The image is the first that works of: the item's feed artwork (a podcast's itunes:image or , an Atom logo), the feed's newest entry's media:thumbnail (a YouTube channel or playlist feed), and the item's thumb. latest is the newest entry's title. feed and thumb are URLs, or asset paths (assets/) for images kept in the site and for fixtures. The image is cropped to size as WebP, so the page never hotlinks a large original. Failures only warn, and a card left without an image shows a monogram, so a dead feed never stops the build. The feed is read on every build, so a scheduled rebuild keeps a channel's latest video current. */ -}} {{- $it := .item -}} {{- $srcs := slice -}} {{- $latest := "" -}} {{- with $it.feed -}} {{- $got := partial "deskbar/feeds/get.html" . -}} {{- $doc := $got.doc -}} {{- range partial "deskbar/feeds/art.html" $doc }}{{ $srcs = $srcs | append . }}{{ end -}} {{- /* feeds list their newest entry first as a rule, but not always, so the dates decide */ -}} {{- $newest := dict -}} {{- $ts := 0 -}} {{- range partial "deskbar/feeds/entries.html" $doc -}} {{- if not (reflect.IsMap .) }}{{ continue }}{{ end -}} {{- $t := 0 -}} {{- with .published | default .pubDate | default .updated | default .date -}} {{- with try (time.AsTime (partial "deskbar/feeds/text.html" .)) }}{{ with .Value }}{{ $t = .Unix }}{{ end }}{{ end -}} {{- end -}} {{- if or (not $newest) (gt $t $ts) }}{{ $newest = . }}{{ $ts = $t }}{{ end -}} {{- end -}} {{- with $newest -}} {{- $latest = partial "deskbar/feeds/text.html" .title | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $th := (.group | default dict).thumbnail | default .thumbnail -}} {{- if reflect.IsSlice $th }}{{ $th = index $th 0 }}{{ end -}} {{- if reflect.IsMap $th }}{{ with index $th "-url" }}{{ $srcs = $srcs | append . }}{{ end }}{{ end -}} {{- end -}} {{- with $got.err }}{{ warnf "Folder %q: feed %s failed: %s" $it.name $it.feed . }}{{ end -}} {{- end -}} {{- with $it.thumb }}{{ $srcs = $srcs | append . }}{{ end -}} {{- $out := dict "latest" $latest -}} {{- range $srcs -}} {{- $img := partial "deskbar/feeds/fetch-image.html" (dict "src" . "spec" (printf "fill %s webp center" $.size)) -}} {{- with $img.src }}{{ $out = merge $out (dict "src" . "width" $img.width "height" $img.height) }}{{ break }}{{ end -}} {{- warnf "Folder %q: no preview image from %s: %s" $it.name . $img.err -}} {{- end -}} {{- return $out -}}
