{{- /* Build-time placeholder art for posts without an image: one object in the style of Haiku's vector icons (dark tone-on-tone outlines, restrained gradients, lit from the top-left). Context: dict "page" and "variant" ("card", a 4:3 tile for Tracker cards and icons: the object on a quiet ground; or "mini", the bare object for list rows, like a Tracker icon). Returns the URL of an SVG file. The ground comes from thumb-seeds.html; the object (emblem) comes from the post's tags and categories. Emblems live in data/deskbar/emblems.yaml. Files are named by their content, so identical mini icons are shared and changed art never hides behind a stale cache. */ -}} {{- $p := .page -}} {{- $mini := eq .variant "mini" -}} {{- $lib := hugo.Data.deskbar.emblems -}} {{- $grounds := $lib.grounds -}} {{- $seed := index (partialCached "deskbar/thumb-seeds.html" site) $p.Path -}} {{- if not $seed -}} {{- $seed = dict "ground" (mod (int (hash.FNV32a $p.Path)) (len $grounds)) -}} {{- end -}} {{- /* Emblem by topic: the first of params.deskbar.thumbRules (defaults in the theme's hugo.toml) with a keyword among the post's tags and categories. */ -}} {{- $terms := slice -}} {{- range slice "tags" "categories" -}} {{- with index $p.Params . -}} {{- if reflect.IsSlice . }}{{ range . }}{{ $terms = $terms | append (lower (string .)) }}{{ end }}{{ else }}{{ $terms = $terms | append (lower (string .)) }}{{ end -}} {{- end -}} {{- end -}} {{- $emblem := "doc" -}} {{- range site.Params.deskbar.thumbRules -}} {{- if and (eq $emblem "doc") (intersect .keywords $terms) }}{{ $emblem = .emblem }}{{ end -}} {{- end -}} {{- $e := index $lib $emblem | default $lib.doc -}} {{- /* Outline widths per variant (see emblems.yaml): the mini icon, drawn at 20-32px, gets heavier outlines and no detail */ -}} {{- $svg := "" -}} {{- if $mini -}} {{- $svg = printf `` $e.defs $e.body -}} {{- else -}} {{- $g := index $grounds $seed.ground -}} {{- $svg = printf `` (index $g 0) (index $g 1) $e.defs $e.body -}} {{- end -}} {{- return (resources.FromString (printf "deskbar/art/%x.svg" (hash.FNV32a $svg)) $svg).RelPermalink -}}
